import pandas as pd
import numpy as np
from sklearn.impute import KNNImputer

# 1. IQR Yöntemiyle Aykırı Değer Budama (Tansiyon Değişkeni)
Q1 = df['kan_basinci'].quantile(0.25)
Q3 = df['kan_basinci'].quantile(0.75)
IQR = Q3 - Q1
df = df[~((df['kan_basinci'] < (Q1 - 1.5 * IQR)) | (df['kan_basinci'] > (Q3 + 1.5 * IQR)))]

# 2. Gelir Düzeyi İçin Güvenli Medyan İmpütasyonu
df['gelir_duzeyi'] = df['gelir_duzeyi'].fillna(df['gelir_duzeyi'].median())
